Ans: echo -e "SERVER_NAME=LocalHost\nPORT=8080\nDEBUG_MODE=true\nMAX_CONN=100" > server.conf sudo mkdir -p /etc/test_config sudo mv server.conf /etc/test_config/server.conf sudo sed -i 's/DEBUG_MODE=true/DEBUG_MODE=false/' /etc/test_config/server.conf sudo chmod 600 /etc/test_config/server.conf
