3. Using terraform create an IAM user that has the administrative access and ec2 full access.

Main.tf

terraform { required_providers { aws = { source  = "hashicorp/aws" version = "~> 6.0" } } }
provider "aws" { region = "ap-southeast-2" }
   1. Create IAM User
resource "aws_iam_user" "devops_user" { name = "terraform-devops-user"
tags = { Name = "Terraform DevOps User" } }
   2. Attach Administrator Access
resource "aws_iam_user_policy_attachment" "admin_access" { user       = aws_iam_user.devops_user.name policy_arn = "arn:aws:iam::aws:policy/AdministratorAccess" }
   3. Attach EC2 Full Access
resource "aws_iam_user_policy_attachment" "ec2_full_access" { user       = aws_iam_user.devops_user.name policy_arn = "arn:aws:iam::aws:policy/AmazonEC2FullAccess" }
   4. Output IAM User Name
output "iam_user_name" { value = aws_iam_user.devops_user.name }
