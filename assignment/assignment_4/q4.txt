4. Using terraform create an ec2 instance and pass a file called provisioners.sh  and the .sh file should contain steps to create a webserver using nginx. 

Provisioners.sh

#!/bin/bash

sudo apt update
sudo apt install -y nginx

sudo systemctl start nginx
sudo systemctl enable nginx

sudo systemctl status nginx –no-pager


main.tf


terraform { required_providers { aws = { source  = "hashicorp/aws" version = "~> 6.0" } } }
provider "aws" { region = "ap-southeast-2" }
   Find the latest Ubuntu 24.04 AMI
data "aws_ami" "ubuntu" { most_recent = true
owners = ["099720109477"]
filter { name   = "name" values = ["ubuntu/images/hvm-ssd-gp3/ubuntu-noble-24.04-amd64-server-*"] }
filter { name   = "virtualization-type" values = ["hvm"] }
filter { name   = "architecture" values = ["x86_64"] } }
   Create Security Group
resource "aws_security_group" "webserver_sg" { name        = "terraform-webserver-sg" description = "Allow SSH and HTTP traffic"
ingress { description = "SSH" from_port   = 22 to_port     = 22 protocol    = "tcp" cidr_blocks = ["0.0.0.0/0"] }
ingress { description = "HTTP" from_port   = 80 to_port     = 80 protocol    = "tcp" cidr_blocks = ["0.0.0.0/0"] }
egress { from_port   = 0 to_port     = 0 protocol    = "-1" cidr_blocks = ["0.0.0.0/0"] }
tags = { Name = "Terraform Webserver SG" } }

   Create EC2 Instance
resource "aws_instance" "webserver" { ami           = data.aws_ami.ubuntu.id instance_type = "t2.micro"
key_name = "custom-key-pair"
vpc_security_group_ids = [ aws_security_group.webserver_sg.id ]
tags = { Name = "Terraform Nginx Webserver" }
   Wait until SSH is available
connection { type        = "ssh" user        = "ubuntu" private_key = file("custom-key-pair.pem") host        = self.public_ip }
   Copy provisioners.sh to the EC2 instance
provisioner "file" { source      = "provisioners.sh" destination = "/home/ubuntu/provisioners.sh" }
   Execute provisioners.sh
provisioner "remote-exec" { inline = [ "chmod +x /home/ubuntu/provisioners.sh", "sudo /home/ubuntu/provisioners.sh" ] } }
   Output public IP
output "public_ip" { value = aws_instance.webserver.public_ip }
   Output website URL
output "website_url" { value = "http://${aws_instance.webserver.public_ip}" }
