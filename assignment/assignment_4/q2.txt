2. Using terraform create an s3 bucket and host a static website in it.

Main.tf
terraform { required_providers { aws = { source  = "hashicorp/aws" version = "~> 6.0" } } }
provider "aws" { region = "ap-southeast-2" }
   1. Create S3 Bucket
resource "aws_s3_bucket" "website_bucket" { bucket = "awneet-terraform-static-website-2026"
tags = { Name = "Terraform Static Website" } }
   2. Disable Block Public Access
resource "aws_s3_bucket_public_access_block" "website_access" { bucket = aws_s3_bucket.website_bucket.id
block_public_acls       = false block_public_policy     = false ignore_public_acls      = false restrict_public_buckets = false }
   3. Configure Static Website
resource "aws_s3_bucket_website_configuration" "website" { bucket = aws_s3_bucket.website_bucket.id
index_document { suffix = "index.html" }
error_document { key = "error.html" } }
   4. Upload index.html
resource "aws_s3_object" "index" { bucket       = aws_s3_bucket.website_bucket.id key          = "index.html" source       = "index.html" content_type = "text/html" }
   5. Upload error.html
resource "aws_s3_object" "error" { bucket       = aws_s3_bucket.website_bucket.id key          = "error.html" source       = "error.html" content_type = "text/html" }
   6. Allow Public Read Access
resource "aws_s3_bucket_policy" "website_policy" { bucket = aws_s3_bucket.website_bucket.id
depends_on = [ aws_s3_bucket_public_access_block.website_access ]
policy = jsonencode({ Version = "2012-10-17"
Statement = [
  {
    Sid       = "PublicReadGetObject"
    Effect    = "Allow"

        Action    = "s3:GetObject"
    Resource  = "${aws_s3_bucket.website_bucket.arn}/*"
  }
]
}) }
   7. Output Website URL
output "website_url" { value = aws_s3_bucket_website_configuration.website.website_endpoint }

index.html
<!DOCTYPE html>
<html>
<head>
    <title>Terraform S3 Website</title>
</head>
<body>
    <h1>Hello from AWS S3!</h1>
    <p>This static website was created and hosted using Terraform.</p>
</body>
</html>

</html>
